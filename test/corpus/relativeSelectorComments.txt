=====================================
Relative selector trailing comments retain rule blocks
=====================================

a {
  > b /* gap */ { color: red; }
  + c // gap
  { color: blue; }
  ~ d /* gap */ { color: green; }
}
.after { background: blue; }

=====================================
Comments before explicit combinators keep both operands
=====================================

b /* c */ > d,
b /* c */ ~ d,
b // c
+ d,
b /* c */ + .d {}
@scope (b /* c */ > d) {}
x:is(b /* c */ + d) {}

---

(stylesheet
  (rule_set
    (selectors
      (child_selector
        (tag_name)
        (comment)
        (tag_name))
      (sibling_selector
        (tag_name)
        (comment)
        (tag_name))
      (adjacent_sibling_selector
        (tag_name)
        (js_comment)
        (tag_name))
      (adjacent_sibling_selector
        (tag_name)
        (comment)
        (class_selector
          (class_name
            (identifier)))))
    (block))
  (scope_statement
    start: (selectors
      (child_selector
        (tag_name)

            (comment)
            (tag_name)))))
    (block)))
